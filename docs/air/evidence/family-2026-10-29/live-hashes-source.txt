import { readdir, readFile, writeFile } from "node:fs/promises";
import { resolve, relative } from "node:path";
import { createHash } from "node:crypto";
const configs = [
  ["cloakyard", "dist", "https://cloakyard.com"],
  ["cloakpdf", "dist", "https://pdf.cloakyard.com"],
  ["cloakdrop", "apps/site/dist", "https://drop.cloakyard.com"],
  ["cloakresume", "dist", "https://resume.cloakyard.com"],
  ["cloakroll", "apps/site/dist", "https://roll.cloakyard.com"],
];
async function download(url) {
  for (let attempt = 1; attempt <= 3; attempt++) {
    try {
      const r = await fetch(url, {
        headers: { "cache-control": "no-cache" },
        signal: AbortSignal.timeout(20000),
      });
      const body = Buffer.from(await r.arrayBuffer());
      if (r.status >= 500 && attempt < 3) continue;
      return { status: r.status, body };
    } catch (e) {
      if (attempt === 3) throw e;
      await new Promise((r) => setTimeout(r, 1000));
    }
  }
}
function stripEdgeDetection(text) {
  return text.replace(
    /<script>\(function\(\)\{function c\(\)\{var b=a\.contentDocument[\s\S]*?<\/script>/g,
    (block) => (block.includes("/cdn-cgi/challenge-platform/scripts/jsd/main.js") ? "" : block),
  );
}
const hash = (b) => createHash("sha256").update(b).digest("hex");
const evidence = [];
async function files(dir) {
  const result = [];
  for (const e of await readdir(dir, { withFileTypes: true })) {
    if (e.name.startsWith(".") || ["_headers", "_redirects"].includes(e.name)) continue;
    const p = resolve(dir, e.name);
    if (e.isDirectory()) result.push(...(await files(p)));
    else result.push(p);
  }
  return result;
}
for (const [repo, sub, origin] of configs) {
  const root = resolve("/Users/sumitsahoo/Developer", repo, sub);
  const list = await files(root);
  for (let i = 0; i < list.length; i += 6) {
    await Promise.all(
      list.slice(i, i + 6).map(async (p) => {
        const path = relative(root, p);
        let route = path === "index.html" ? "" : path.replace(/\/index.html$/, "/");
        const r = await download(origin + "/" + route);
        const actual = hash(r.body),
          expected = hash(await readFile(p));
        const normalized = path.endsWith(".html")
          ? hash(Buffer.from(stripEdgeDetection(r.body.toString("utf8"))))
          : actual;
        evidence.push({
          repo,
          path,
          status: r.status,
          expected,
          actual,
          normalized,
          edgeDetection: normalized !== actual,
          match: expected === normalized,
        });
      }),
    );
  }
  console.log(
    repo,
    list.length,
    "assets",
    evidence.filter((x) => x.repo === repo && !x.match).length,
    "mismatches",
  );
}
await writeFile(
  "/tmp/air-family-2026-10-29/live-hashes.json",
  JSON.stringify(evidence, null, 2) + "\n",
);
if (evidence.some((x) => !x.match || ![200, 404].includes(x.status))) process.exitCode = 1;
